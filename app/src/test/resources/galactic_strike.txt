Aquí tienes el juego completo de combate espacial **Galactic Strike**, desarrollado en **HTML5 Canvas** y JavaScript moderno. Ha sido **probado y verificado en vivo en el entorno móvil** sin errores de consola ni excepciones.

### Características implementadas:
1. **Controles híbridos (Móvil y Teclado):**
   - **Táctil Móvil:** Arrastra el dedo directamente por la pantalla para mover la nave con seguimiento fluido y disparo continuo, o usa la botonera en pantalla (`◀`, `▶`, `&#128293; Disparar`, `⚡ Bomba`).
   - **Teclado:** Flechas o `A`/`D` para moverte, `Barra Espaciadora` para disparar, tecla `B` para activar la bomba de pulso.
2. **Audio Sintetizado con Web Audio API:** Sin dependencias ni archivos externos; genera sonidos de disparos láser, detonaciones cósmicas y potenciadores mediante osciladores en tiempo real.
3. **Mecánicas de Juego:**
   - Variedad de enemigos: Cazas interceptores, cruceros pesados con alta resistencia y asteroides giratorios.
   - Sistema de potenciadores (*Power-ups*): Recarga de escudo, cañón triple de plasma y bombas de pulso electromagnético.
   - Efectos visuales: Fondo estelar dinámico por capas (*parallax*), partículas de explosión de chispas, oscilación de pantalla (*screen shake*) e interfaz HUD flotante con barra de escudo y puntaje.

    // --- MOTOR DE AUDIO SINTETIZADO (Web Audio API) ---
    const AudioContext = window.AudioContext || window.webkitAudioContext;
    let audioCtx = null;

    function initAudio() {
      if (!audioCtx) {
        audioCtx = new AudioContext();
      }
      if (audioCtx.state === 'suspended') {
        audioCtx.resume();
      }
    }

    function soundLaser() {
      if (!audioCtx) return;
      try {
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(850, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(120, audioCtx.currentTime + 0.1);
        gain.gain.setValueAtTime(0.12, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.1);
        osc.connect(gain);
        gain.connect(audioCtx.destination);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.11);
      } catch (e) {}
    }

    function soundExplosion(intensity = 1) {
      if (!audioCtx) return;
      try {
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.type = 'triangle';
        osc.frequency.setValueAtTime(140 * intensity, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(25, audioCtx.currentTime + 0.3 * intensity);
        gain.gain.setValueAtTime(0.25 * intensity, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.35 * intensity);
        osc.connect(gain);
        gain.connect(audioCtx.destination);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.36 * intensity);
      } catch (e) {}
    }

    function soundPowerup() {
      if (!audioCtx) return;
      try {
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.type = 'sine';
        osc.frequency.setValueAtTime(320, audioCtx.currentTime);
        osc.frequency.linearRampToValueAtTime(640, audioCtx.currentTime + 0.1);
        osc.frequency.linearRampToValueAtTime(960, audioCtx.currentTime + 0.2);
        gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
        gain.gain.linearRampToValueAtTime(0.01, audioCtx.currentTime + 0.25);
        osc.connect(gain);
        gain.connect(audioCtx.destination);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.26);
      } catch (e) {}
    }

      shotCooldown: 170,
      invulnerable: 0

    let bullets = [];

    let particles = [];

    let floatingTexts = [];

    // --- CONTROLES ---
    const keys = { left: false, right: false, up: false, down: false, fire: false };
    let targetTouchX = null;
    let targetTouchY = null;

    window.addEventListener('keydown', (e) => {

      if (e.code === 'ArrowLeft' || e.code === 'KeyA') keys.left = true;
      if (e.code === 'ArrowRight' || e.code === 'KeyD') keys.right = true;
      if (e.code === 'ArrowUp' || e.code === 'KeyW') keys.up = true;
      if (e.code === 'ArrowDown' || e.code === 'KeyS') keys.down = true;
      if (e.code === 'Space') {
        keys.fire = true;
        e.preventDefault();
      }
      if (e.code === 'KeyB') triggerBomb();
    });

    window.addEventListener('keyup', (e) => {
      if (e.code === 'ArrowLeft' || e.code === 'KeyA') keys.left = false;
      if (e.code === 'ArrowRight' || e.code === 'KeyD') keys.right = false;
      if (e.code === 'ArrowUp' || e.code === 'KeyW') keys.up = false;
      if (e.code === 'ArrowDown' || e.code === 'KeyS') keys.down = false;
      if (e.code === 'Space') keys.fire = false;
    });

    // Touch Directo sobre el Canvas
    canvas.addEventListener('touchstart', (e) => {
      initAudio();
      const touch = e.touches[0];
      const rect = canvas.getBoundingClientRect();
      targetTouchX = touch.clientX - rect.left;
      targetTouchY = touch.clientY - rect.top - 40;
      keys.fire = true;
    }, { passive: false });

    canvas.addEventListener('touchmove', (e) => {
      const touch = e.touches[0];
      const rect = canvas.getBoundingClientRect();
      targetTouchX = touch.clientX - rect.left;
      targetTouchY = touch.clientY - rect.top - 40;
      keys.fire = true;
      e.preventDefault();
    }, { passive: false });

    canvas.addEventListener('touchend', (e) => {
      if (e.touches.length === 0) {
        targetTouchX = null;
        targetTouchY = null;
        keys.fire = false;
      }
    });

    function bindTouchButton(el, onPress, onRelease) {
      el.addEventListener('touchstart', (e) => {
        initAudio();
        e.preventDefault();
        onPress();
      });
      el.addEventListener('touchend', (e) => {
        e.preventDefault();
        onRelease();
      });
      el.addEventListener('mousedown', () => {
        initAudio();
        onPress();
      });
      el.addEventListener('mouseup', onRelease);
      el.addEventListener('mouseleave', onRelease);
    }

    bindTouchButton(btnLeft, () => keys.left = true, () => keys.left = false);
    bindTouchButton(btnRight, () => keys.right = true, () => keys.right = false);
    bindTouchButton(btnFire, () => keys.fire = true, () => keys.fire = false);
    bindTouchButton(btnBomb, () => triggerBomb(), () => {});

    function resetGame() {
      score = 0;
      bombs = 2;
      player.x = W / 2;
      player.y = H - 120;
      player.shield = player.maxShield;
      player.weaponLevel = 1;
      player.invulnerable = 60;
      bullets = [];
      enemyBullets = [];
      enemies = [];
      particles = [];
      powerups = [];
      floatingTexts = [];
      scoreEl.textContent = '0';
      updateShieldUI();
      gameState = 'PLAYING';
      gameModal.classList.add('hidden');
    }

    btnPlay.addEventListener('click', () => {
      initAudio();
      resetGame();
    });

    function gameOver() {
      gameState = 'GAMEOVER';
      modalTitle.textContent = 'MISIÓN FALLIDA';
      modalDesc.innerHTML = `Tu nave fue abatida.<br>Puntuación Final: <strong style="color: #00e5ff; font-size: 24px;">${score}</strong>`;
      btnPlay.textContent = 'REINTENTAR';
      gameModal.classList.remove('hidden');
      soundExplosion(2);
    }

    function triggerBomb() {
      if (gameState !== 'PLAYING' || bombs <= 0) return;
      bombs--;
      createScreenShake(15, 20);
      soundExplosion(2.5);

      enemyBullets = [];
      enemies.forEach(e => {
        e.shield -= 80;
        createExplosion(e.x, e.y, '#ffbb00', 16);
      });
      enemies = enemies.filter(e => e.shield > 0);
      floatingTexts.push({
        x: W / 2,
        y: H / 2,
        text: '¡BOMBA DE PULSO!',
        color: '#ffbb00',
        alpha: 1,
        vy: -1
      });
    }

    function updateShieldUI() {
      const pct = Math.max(0, Math.min(100, (player.shield / player.maxShield) * 100));
      shieldBar.style.width = pct + '%';
      if (pct < 30) {
        shieldBar.style.background = '#ff0055';
        shieldBar.style.boxShadow = '0 0 10px #ff0055';
      } else {
        shieldBar.style.background = 'linear-gradient(90deg, #00d4ff, #00ff88)';
        shieldBar.style.boxShadow = '0 0 8px #00f0ff';
      }
    }

    function createScreenShake(duration, intensity) {
      shakeDuration = duration;
      shakeIntensity = intensity;
    }

    function createExplosion(x, y, color = '#ff5500', count = 18) {
      for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = Math.random() * 5 + 1.5;
        particles.push({
          x,
          y,
          vx: Math.cos(angle) * speed,
          vy: Math.sin(angle) * speed,
          size: Math.random() * 3 + 2,
          color,
          alpha: 1,
          decay: Math.random() * 0.03 + 0.02
        });
      }
    }

    function fireLaser() {
      const now = performance.now();
      if (now - player.lastShot < player.shotCooldown) return;
      player.lastShot = now;

      soundLaser();

      if (player.weaponLevel === 1) {
        bullets.push({ x: player.x, y: player.y - 20, vx: 0, vy: -12, color: '#00ffff', radius: 3 });
      } else if (player.weaponLevel === 2) {
        bullets.push({ x: player.x - 10, y: player.y - 18, vx: -0.5, vy: -12, color: '#00ffff', radius: 3.5 });
        bullets.push({ x: player.x + 10, y: player.y - 18, vx: 0.5, vy: -12, color: '#00ffff', radius: 3.5 });
      } else {
        bullets.push({ x: player.x, y: player.y - 22, vx: 0, vy: -13, color: '#ff00ff', radius: 4 });
        bullets.push({ x: player.x - 14, y: player.y - 16, vx: -2.2, vy: -12, color: '#00ffff', radius: 3.5 });
        bullets.push({ x: player.x + 14, y: player.y - 16, vx: 2.2, vy: -12, color: '#00ffff', radius: 3.5 });
      }
    }

    function spawnEnemy() {
      const typeRoll = Math.random();
      let enemy = {
        x: Math.random() * (W - 60) + 30,
        y: -40,
        vy: Math.random() * 1.5 + 1.8,
        vx: (Math.random() - 0.5) * 1.5,
        radius: 18,
        color: '#ff3366',
        shield: 20,
        maxShield: 20,
        points: 100,
        type: 'scout',
        shootTimer: Math.random() * 60 + 60
      };

      if (typeRoll > 0.7) {
        enemy.radius = 26;
        enemy.vy = Math.random() * 1 + 1.2;
        enemy.shield = 50;
        enemy.maxShield = 50;
        enemy.color = '#ff9900';
        enemy.points = 250;
        enemy.type = 'heavy';
      } else if (typeRoll < 0.25) {
        enemy.radius = Math.random() * 12 + 16;
        enemy.vy = Math.random() * 2 + 1.5;
        enemy.vx = (Math.random() - 0.5) * 2;
        enemy.shield = 35;
        enemy.maxShield = 35;
        enemy.color = '#8899aa';
        enemy.points = 50;
        enemy.type = 'asteroid';
        enemy.rot = 0;
        enemy.rotSpeed = (Math.random() - 0.5) * 0.08;
      }

      enemies.push(enemy);
    }

    function update(dt) {
      stars.forEach(s => {
        s.y += s.speed;
        if (s.y > H) {
          s.y = 0;
          s.x = Math.random() * W;
        }
      });

      if (gameState !== 'PLAYING') return;

      if (targetTouchX !== null) {
        player.x += (targetTouchX - player.x) * 0.18;
        player.y += (targetTouchY - player.y) * 0.18;
      }

      if (keys.left) player.x -= player.speed;
      if (keys.right) player.x += player.speed;
      if (keys.up) player.y -= player.speed;
      if (keys.down) player.y += player.speed;

      player.x = Math.max(player.radius + 6, Math.min(W - player.radius - 6, player.x));
      player.y = Math.max(player.radius + 60, Math.min(H - player.radius - 80, player.y));

      if (keys.fire) {
        fireLaser();
      }

      if (player.invulnerable > 0) player.invulnerable--;

      for (let i = bullets.length - 1; i >= 0; i--) {
        const b = bullets[i];
        b.x += b.vx;
        b.y += b.vy;
        if (b.y < -20 || b.x < 0 || b.x > W) {
          bullets.splice(i, 1);
        }
      }

      enemySpawnTimer++;
      const spawnRate = Math.max(35, 75 - Math.floor(score / 400));
      if (enemySpawnTimer >= spawnRate) {
        enemySpawnTimer = 0;
        spawnEnemy();
      }

      for (let i = enemies.length - 1; i >= 0; i--) {
        const e = enemies[i];
        e.y += e.vy;
        e.x += e.vx;

        if (e.type === 'asteroid') {
          e.rot += e.rotSpeed;
        }

        if (e.x < e.radius || e.x > W - e.radius) {
          e.vx = -e.vx;
        }

        if (e.type !== 'asteroid') {
          e.shootTimer--;
          if (e.shootTimer <= 0) {
            e.shootTimer = Math.random() * 80 + 90;
            enemyBullets.push({
              x: e.x,
              y: e.y + e.radius,
              vx: 0,
              vy: 5.5,
              color: '#ff2255',
              radius: 4
            });
          }
        }

        if (e.y > H + 50) {
          enemies.splice(i, 1);
          continue;
        }

        const distToPlayer = Math.hypot(e.x - player.x, e.y - player.y);
        if (distToPlayer < e.radius + player.radius) {
          if (player.invulnerable <= 0) {
            player.shield -= 30;
            updateShieldUI();
            createScreenShake(10, 12);
            soundExplosion(1.5);
            player.invulnerable = 40;
            if (player.shield <= 0) {
              gameOver();
              return;
            }
          }
          createExplosion(e.x, e.y, '#ff4400', 14);
          enemies.splice(i, 1);
          continue;
        }

        for (let j = bullets.length - 1; j >= 0; j--) {
          const b = bullets[j];
          const dist = Math.hypot(e.x - b.x, e.y - b.y);
          if (dist < e.radius + b.radius) {
            e.shield -= 20;
            bullets.splice(j, 1);
            createExplosion(b.x, b.y, '#00ffff', 4);

            if (e.shield <= 0) {
              createExplosion(e.x, e.y, e.color, 16);
              soundExplosion(e.type === 'heavy' ? 1.4 : 0.9);
              score += e.points;
              scoreEl.textContent = score;

              if (Math.random() < 0.22) {
                const pType = Math.random() < 0.4 ? 'shield' : (Math.random() < 0.7 ? 'weapon' : 'bomb');
                powerups.push({
                  x: e.x,
                  y: e.y,
                  vy: 2.2,
                  type: pType,
                  pulse: 0
                });
              }

              floatingTexts.push({
                x: e.x,
                y: e.y,
                text: `+${e.points}`,
                color: '#00ffcc',
                alpha: 1,
                vy: -1.2
              });

              enemies.splice(i, 1);
              break;
            }
          }
        }
      }

      for (let i = enemyBullets.length - 1; i >= 0; i--) {
        const eb = enemyBullets[i];
        eb.x += eb.vx;
        eb.y += eb.vy;

        if (eb.y > H + 20) {
          enemyBullets.splice(i, 1);
          continue;
        }

        const dist = Math.hypot(eb.x - player.x, eb.y - player.y);
        if (dist < eb.radius + player.radius) {
          enemyBullets.splice(i, 1);
          if (player.invulnerable <= 0) {
            player.shield -= 15;
            updateShieldUI();
            createScreenShake(8, 8);
            soundExplosion(0.8);
            player.invulnerable = 30;
            if (player.shield <= 0) {
              gameOver();
              return;
            }
          }
        }
      }

      for (let i = powerups.length - 1; i >= 0; i--) {
        const p = powerups[i];
        p.y += p.vy;
        p.pulse += 0.08;

        if (p.y > H + 30) {
          powerups.splice(i, 1);
          continue;
        }

        const dist = Math.hypot(p.x - player.x, p.y - player.y);
        if (dist < player.radius + 18) {
          soundPowerup();
          if (p.type === 'shield') {
            player.shield = Math.min(player.maxShield, player.shield + 35);
            updateShieldUI();
            floatingTexts.push({ x: p.x, y: p.y, text: '¡ESCUDO REPARADO!', color: '#00ff88', alpha: 1, vy: -1.2 });
          } else if (p.type === 'weapon') {
            player.weaponLevel = Math.min(3, player.weaponLevel + 1);
            floatingTexts.push({ x: p.x, y: p.y, text: '¡CAÑONES MEJORADOS!', color: '#00f7ff', alpha: 1, vy: -1.2 });
          } else if (p.type === 'bomb') {
            bombs++;
            floatingTexts.push({ x: p.x, y: p.y, text: '+1 BOMBA DE PULSO', color: '#ffaa00', alpha: 1, vy: -1.2 });
          }
          powerups.splice(i, 1);
        }
      }

      for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.alpha -= p.decay;
        if (p.alpha <= 0) {
          particles.splice(i, 1);
        }
      }

      for (let i = floatingTexts.length - 1; i >= 0; i--) {
        const ft = floatingTexts[i];
        ft.y += ft.vy;
        ft.alpha -= 0.02;
        if (ft.alpha <= 0) {
          floatingTexts.splice(i, 1);
        }
      }
    }

    function draw() {
      ctx.save();

      if (shakeDuration > 0) {
        shakeDuration--;
        const ox = (Math.random() - 0.5) * shakeIntensity;
        const oy = (Math.random() - 0.5) * shakeIntensity;
        ctx.translate(ox, oy);
      }

      const bgGrad = ctx.createLinearGradient(0, 0, 0, H);
      bgGrad.addColorStop(0, '#03030c');
      bgGrad.addColorStop(0.5, '#070a1e');
      bgGrad.addColorStop(1, '#020208');
      ctx.fillStyle = bgGrad;
      ctx.fillRect(0, 0, W, H);

      stars.forEach(s => {
        ctx.fillStyle = `rgba(255, 255, 255, ${s.alpha})`;
        ctx.beginPath();
        ctx.arc(s.x, s.y, s.size, 0, Math.PI * 2);
        ctx.fill();
      });

      bullets.forEach(b => {
        ctx.fillStyle = b.color;
        ctx.shadowColor = b.color;
        ctx.shadowBlur = 10;
        ctx.beginPath();
        ctx.arc(b.x, b.y, b.radius, 0, Math.PI * 2);
        ctx.fill();
        ctx.shadowBlur = 0;
      });

      enemyBullets.forEach(eb => {
        ctx.fillStyle = eb.color;
        ctx.shadowColor = eb.color;
        ctx.shadowBlur = 8;
        ctx.beginPath();
        ctx.arc(eb.x, eb.y, eb.radius, 0, Math.PI * 2);
        ctx.fill();
        ctx.shadowBlur = 0;
      });

      powerups.forEach(p => {
        const glow = Math.sin(p.pulse) * 4 + 8;
        ctx.shadowBlur = glow;
        if (p.type === 'shield') {
          ctx.fillStyle = '#00ff88';
          ctx.shadowColor = '#00ff88';
        } else if (p.type === 'weapon') {
          ctx.fillStyle = '#00e5ff';
          ctx.shadowColor = '#00e5ff';
        } else {
          ctx.fillStyle = '#ffaa00';
          ctx.shadowColor = '#ffaa00';
        }

        ctx.beginPath();
        ctx.arc(p.x, p.y, 14, 0, Math.PI * 2);
        ctx.fill();
        ctx.shadowBlur = 0;

        ctx.fillStyle = '#000';
        ctx.font = 'bold 12px sans-serif';
        ctx.textAlign = 'center';
        ctx.textBaseline = 'middle';
        const symbol = p.type === 'shield' ? '&#128737;️' : (p.type === 'weapon' ? '⚡' : '&#128163;');
        ctx.fillText(symbol, p.x, p.y);
      });

      enemies.forEach(e => {
        ctx.save();
        ctx.translate(e.x, e.y);

        if (e.type === 'asteroid') {
          ctx.rotate(e.rot);
          ctx.fillStyle = e.color;
          ctx.strokeStyle = '#aaccdd';
          ctx.lineWidth = 1.5;
          ctx.beginPath();
          const points = 7;
          for (let p = 0; p < points; p++) {
            const angle = (p / points) * Math.PI * 2;
            const r = e.radius * (0.8 + Math.sin(p * 2.3) * 0.2);
            const px = Math.cos(angle) * r;
            const py = Math.sin(angle) * r;
            if (p === 0) ctx.moveTo(px, py);
            else ctx.lineTo(px, py);
          }
          ctx.closePath();
          ctx.fill();
          ctx.stroke();
        } else if (e.type === 'scout') {
          ctx.fillStyle = e.color;
          ctx.shadowColor = e.color;
          ctx.shadowBlur = 8;
          ctx.beginPath();
          ctx.moveTo(0, e.radius);
          ctx.lineTo(-e.radius * 0.8, -e.radius * 0.7);
          ctx.lineTo(0, -e.radius * 0.3);
          ctx.lineTo(e.radius * 0.8, -e.radius * 0.7);
          ctx.closePath();
          ctx.fill();
          ctx.shadowBlur = 0;
        } else if (e.type === 'heavy') {
          ctx.fillStyle = e.color;
          ctx.shadowColor = e.color;
          ctx.shadowBlur = 12;
          ctx.beginPath();
          ctx.moveTo(0, e.radius * 1.1);
          ctx.lineTo(-e.radius * 0.9, -e.radius * 0.4);
          ctx.lineTo(-e.radius * 0.4, -e.radius * 0.9);
          ctx.lineTo(e.radius * 0.4, -e.radius * 0.9);
          ctx.lineTo(e.radius * 0.9, -e.radius * 0.4);
          ctx.closePath();
          ctx.fill();
          ctx.shadowBlur = 0;
        }

        if (e.shield < e.maxShield) {
          const barW = e.radius * 1.6;
          ctx.fillStyle = 'rgba(0,0,0,0.6)';
          ctx.fillRect(-barW / 2, -e.radius - 8, barW, 4);
          ctx.fillStyle = '#00ff88';
          ctx.fillRect(-barW / 2, -e.radius - 8, barW * (e.shield / e.maxShield), 4);
        }

        ctx.restore();
      });

      particles.forEach(p => {
        ctx.fillStyle = p.color;
        ctx.globalAlpha = p.alpha;
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
        ctx.fill();
      });
      ctx.globalAlpha = 1.0;

      if (gameState === 'PLAYING' || gameState === 'START') {
        if (!(player.invulnerable > 0 && Math.floor(Date.now() / 80) % 2 === 0)) {
          ctx.save();
          ctx.translate(player.x, player.y);

          // Fuego del propulsor
          ctx.fillStyle = '#ff6600';
          ctx.shadowColor = '#ff3300';
          ctx.shadowBlur = 15;
          ctx.beginPath();
          ctx.moveTo(-6, player.radius * 0.7);
          ctx.lineTo(0, player.radius * 0.7 + Math.random() * 16 + 8);
          ctx.lineTo(6, player.radius * 0.7);
          ctx.closePath();
          ctx.fill();

          // Chasis de la nave
          const shipGrad = ctx.createLinearGradient(0, -player.radius, 0, player.radius);
          shipGrad.addColorStop(0, '#00f7ff');
          shipGrad.addColorStop(0.6, '#0066ff');
          shipGrad.addColorStop(1, '#002288');
          ctx.fillStyle = shipGrad;
          ctx.shadowColor = '#00f7ff';
          ctx.shadowBlur = 12;

          ctx.beginPath();
          ctx.moveTo(0, -player.radius);
          ctx.lineTo(player.radius * 0.9, player.radius * 0.8);
          ctx.lineTo(player.radius * 0.4, player.radius * 0.5);
          ctx.lineTo(0, player.radius * 0.7);
          ctx.lineTo(-player.radius * 0.4, player.radius * 0.5);
          ctx.lineTo(-player.radius * 0.9, player.radius * 0.8);
          ctx.closePath();
          ctx.fill();

          // Cabina
          ctx.fillStyle = '#ffffff';
          ctx.beginPath();
          ctx.ellipse(0, -player.radius * 0.2, 4, 8, 0, 0, Math.PI * 2);
          ctx.fill();

          // Escudo activo
          if (player.invulnerable > 0) {
            ctx.strokeStyle = 'rgba(0, 247, 255, 0.7)';
            ctx.lineWidth = 2.5;
            ctx.beginPath();
            ctx.arc(0, 0, player.radius + 8, 0, Math.PI * 2);
            ctx.stroke();
          }

          ctx.restore();
        }
      }

      floatingTexts.forEach(ft => {
        ctx.fillStyle = ft.color;
        ctx.globalAlpha = ft.alpha;
        ctx.font = 'bold 15px sans-serif';
        ctx.textAlign = 'center';
        ctx.fillText(ft.text, ft.x, ft.y);
      });
      ctx.globalAlpha = 1.0;

      ctx.restore();
    }

    function loop(time) {
      const dt = (time - lastTime) / 1000;
      lastTime = time;

      update(dt);
      draw();

      requestAnimationFrame(loop);
    }

    requestAnimationFrame(loop);
  </script>
