El juego espacial arcade **Galaxy Striker** ha sido probado y verificado con éxito en el motor WebView del dispositivo.

### Características implementadas:
- **Controles táctiles y de teclado duales**:
  - **Móvil / Pantalla táctil**: Puedes arrastrar la nave directamente con el dedo o usar la cruceta y botones táctiles en pantalla (*FUEGO* y *BOMBA*).
  - **Teclado**: Flechas o `A`/`D`/`W`/`S` para desplazarse, `Espacio`/`J`/`Z` para disparar y `B`/`K`/`Shift` para lanzar la bomba EMP.
- **Sintetizador de audio procedural (Web Audio API)**: Láseres, disparos enemigos, explosiones, recogida de objetos y detonación EMP generados en tiempo real sin dependencias externas.
- **Variedad de enemigos**:
  - **Drones de asalto**: Desplazamiento oscilatorio y fuego frontal.
  - **Asteroides espaciales**: Se fragmentan en trozos más pequeños al ser destruidos.
  - **Cruceros pesados**: Disparan ráfagas dobles y cuentan con blindaje.
  - **Jefe Destructor Nova**: Aparece con barra de vida dedicada y patrones de ráfaga triple.
- **Sistema de Power-Ups**:
  - **W**: Mejora de armas a láser múltiple (hasta 4 cañones en abanico).
  - **+**: Reparación de escudos e integridad de la nave.
  - **B**: Bombas EMP de área que eliminan proyectiles hostiles e infligen daño masivo.
- **Efectos visuales**: Sacudida de pantalla (*screen shake*), onda de choque EMP expansiva, estelas de partículas de propulsión y guardado de puntuación máxima persistente en `localStorage`.

(() => {
  const canvas = document.getElementById('gameCanvas');

  // --- AUDIO SYNTHESIZER (Web Audio API) ---
  const AudioEngine = {
    ctx: null,
    init() {
      if (!this.ctx) {
        const AudioCtx = window.AudioContext || window.webkitAudioContext;
        if (AudioCtx) this.ctx = new AudioCtx();
      }
      if (this.ctx && this.ctx.state === 'suspended') {
        this.ctx.resume();
      }
    },
    laser() {
      if (!this.ctx) return;
      try {
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(750, this.ctx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(120, this.ctx.currentTime + 0.1);
        gain.gain.setValueAtTime(0.12, this.ctx.currentTime);
        gain.gain.linearRampToValueAtTime(0.01, this.ctx.currentTime + 0.1);
        osc.connect(gain);
        gain.connect(this.ctx.destination);
        osc.start();
        osc.stop(this.ctx.currentTime + 0.1);
      } catch (e) {}
    },
    enemyLaser() {
      if (!this.ctx) return;
      try {
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.type = 'square';
        osc.frequency.setValueAtTime(320, this.ctx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(80, this.ctx.currentTime + 0.12);
        gain.gain.setValueAtTime(0.08, this.ctx.currentTime);
        gain.gain.linearRampToValueAtTime(0.01, this.ctx.currentTime + 0.12);
        osc.connect(gain);
        gain.connect(this.ctx.destination);
        osc.start();
        osc.stop(this.ctx.currentTime + 0.12);
      } catch (e) {}
    },
    explode(big = false) {
      if (!this.ctx) return;
      try {
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.type = 'triangle';
        osc.frequency.setValueAtTime(big ? 150 : 260, this.ctx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(20, this.ctx.currentTime + (big ? 0.4 : 0.2));
        gain.gain.setValueAtTime(big ? 0.3 : 0.15, this.ctx.currentTime);
        gain.gain.linearRampToValueAtTime(0.01, this.ctx.currentTime + (big ? 0.4 : 0.2));
        osc.connect(gain);
        gain.connect(this.ctx.destination);
        osc.start();
        osc.stop(this.ctx.currentTime + (big ? 0.4 : 0.2));
      } catch (e) {}
    },
    powerup() {
      if (!this.ctx) return;
      try {
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.type = 'sine';
        osc.frequency.setValueAtTime(300, this.ctx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(900, this.ctx.currentTime + 0.2);

        osc.start();

        osc.stop(this.ctx.currentTime + 0.6);

    update() {
      let moveX = 0;
      let moveY = 0;
      if (keys.left) moveX -= 1;
      if (keys.right) moveX += 1;
      if (keys.up) moveY -= 1;

        this.y += this.vy;
        this.tilt = this.vx * 0.04;

  // --- PARTÍCULAS Y TEXTOS FLOTANTES ---
  class Particle {
    constructor(x, y, vx, vy, color, life = 20, size = 3) {
      this.x = x;
      this.y = y;
      this.vx = vx;
      this.vy = vy;
      this.color = color;
      this.maxLife = life;
      this.life = life;
      this.size = size;
    }
    update() {
      this.x += this.vx;
      this.y += this.vy;
      this.life--;
    }
    draw() {
      const alpha = Math.max(0, this.life / this.maxLife);
      ctx.save();
      ctx.globalAlpha = alpha;
      ctx.fillStyle = this.color;
      ctx.beginPath();
      ctx.arc(this.x, this.y, this.size * alpha, 0, Math.PI * 2);
      ctx.fill();
      ctx.restore();
    }
  }

  class FloatingText {
    constructor(x, y, text, color = '#ffd600') {
      this.x = x;
      this.y = y;
      this.text = text;
      this.color = color;
      this.life = 40;

      this.life--;

      ctx.fillStyle = this.color;
      ctx.font = 'bold 16px monospace';
      ctx.textAlign = 'center';
      ctx.fillText(this.text, this.x, this.y);
      ctx.restore();

  function createExplosion(x, y, color = '#ff8800', count = 18) {
    for (let i = 0; i < count; i++) {
      const angle = Math.random() * Math.PI * 2;
      const speed = 1.5 + Math.random() * 5.5;
      particles.push(new Particle(
        x,
        y,
        Math.cos(angle) * speed,
        Math.sin(angle) * speed,
        color,
        25 + Math.random() * 15,
        2.5 + Math.random() * 3
      ));
    }
  }

      ctx.rotate(this.rot);

      ctx.fillStyle = '#4a5369';
      ctx.strokeStyle = '#8592ad';
      ctx.lineWidth = 2;

      ctx.beginPath();
      for (let i = 0; i < this.pointsCount; i++) {
        const angle = (i / this.pointsCount) * Math.PI * 2;
        const r = this.radius * this.offsets[i];
        const px = Math.cos(angle) * r;
        const py = Math.sin(angle) * r;
        if (i === 0) ctx.moveTo(px, py);
        else ctx.lineTo(px, py);
      }
      ctx.closePath();
      ctx.fill();
      ctx.stroke();

      ctx.restore();
    }
  }

  class Cruiser {
    constructor(x, y) {
      this.x = x;
      this.y = y;
      this.w = 56;
      this.h = 48;
      this.hp = 85;
      this.maxHp = 85;
      this.points = 250;
      this.vy = 1.5;
      this.shootCooldown = 45;
    }
    update() {
      this.y += this.vy;
      this.shootCooldown--;
      if (this.shootCooldown <= 0 && this.y < H - 180) {
        AudioEngine.enemyLaser();
        enemyProjectiles.push(new EnemyLaser(this.x - 14, this.y + 20, -0.6, 6));
        enemyProjectiles.push(new EnemyLaser(this.x + 14, this.y + 20, 0.6, 6));
        this.shootCooldown = 75;
      }
    }
    draw() {
      ctx.save();
      ctx.translate(this.x, this.y);
      ctx.shadowBlur = 12;
      ctx.shadowColor = '#ff6a00';

      ctx.fillStyle = '#2b1b36';
      ctx.strokeStyle = '#ff7700';
      ctx.lineWidth = 2;

      ctx.beginPath();
      ctx.moveTo(0, 24);
      ctx.lineTo(26, -10);
      ctx.lineTo(20, -22);
      ctx.lineTo(0, -14);
      ctx.lineTo(-20, -22);
      ctx.lineTo(-26, -10);
      ctx.closePath();
      ctx.fill();
      ctx.stroke();

      ctx.fillStyle = '#ffaa00';
      ctx.fillRect(-6, -4, 12, 10);

        ctx.fillStyle = 'rgba(0,0,0,0.5)';
        ctx.fillRect(-22, -32, 44, 4);
        ctx.fillStyle = '#00ff88';
        ctx.fillRect(-22, -32, 44 * Math.max(0, this.hp / this.maxHp), 4);
      }

      ctx.restore();
    }
  }

  class Boss {

      this.y = -100;

      this.w = 140;
      this.h = 90;
      this.hp = 650;
      this.maxHp = 650;
      this.points = 1500;
      this.speedX = 2.2;
      this.shootTimer = 0;
      this.isBoss = true;

    update() {
      if (this.y < this.targetY) {
        this.y += 1.8;
      } else {
        this.x += this.speedX;
        if (this.x < 110 || this.x > W - 110) {
          this.speedX *= -1;
        }
      }

      this.shootTimer++;
      if (this.shootTimer % 45 === 0) {
        AudioEngine.enemyLaser();
        enemyProjectiles.push(new EnemyLaser(this.x - 35, this.y + 35, -2, 5.5));
        enemyProjectiles.push(new EnemyLaser(this.x, this.y + 40, 0, 6));
        enemyProjectiles.push(new EnemyLaser(this.x + 35, this.y + 35, 2, 5.5));
      }
    }
    draw() {
      ctx.save();
      ctx.translate(this.x, this.y);
      ctx.shadowBlur = 18;
      ctx.shadowColor = '#ff0055';

      ctx.fillStyle = '#1c0f24';
      ctx.strokeStyle = '#ff0055';
      ctx.lineWidth = 3;

      ctx.beginPath();

      ctx.lineTo(65, 0);
      ctx.lineTo(55, -40);
      ctx.lineTo(0, -25);
      ctx.lineTo(-55, -40);
      ctx.lineTo(-65, 0);

      ctx.fill();
      ctx.stroke();

  function getCanvasCoords(e) {
    const rect = canvas.getBoundingClientRect();
    const scaleX = canvas.width / rect.width;
    const scaleY = canvas.height / rect.height;
    return {
      x: (e.clientX - rect.left) * scaleX,
      y: (e.clientY - rect.top) * scaleY
    };
  }

    AudioEngine.init();
    if (state !== 'PLAYING') {
      startGame();
      return;

    isTouchingCanvas = true;
    touchTarget = getCanvasCoords(e);
  });

  canvas.addEventListener('pointermove', (e) => {
    if (isTouchingCanvas && state === 'PLAYING') {
      touchTarget = getCanvasCoords(e);
    }
  });

  const stopCanvasPointer = () => {
    isTouchingCanvas = false;
    touchTarget = null;

  // --- CICLO DEL JUEGO ---
  function startGame() {
    AudioEngine.init();
    player.reset();
    projectiles = [];
    enemyProjectiles = [];
    enemies = [];
    particles = [];
    powerups = [];
    floatingTexts = [];
    score = 0;
    bossActive = false;
    nextBossScore = 1500;
    spawnTimer = 0;
    state = 'PLAYING';
  }

  function endGame() {
    state = 'GAMEOVER';
    if (score > highScore) {
      highScore = score;
      try {
        localStorage.setItem('galaxy_striker_hiscore', highScore.toString());
      } catch (e) {}
    }
  }

  function updateSpawner() {
    if (bossActive) return;

    if (score >= nextBossScore) {
      bossActive = true;
      enemies.push(new Boss());
      nextBossScore += 2500;
      floatingTexts.push(new FloatingText(W / 2, 220, '¡JEFE DETECTADO!', '#ff0055'));
      return;
    }

    spawnTimer++;
    if (spawnTimer >= 48) {
      spawnTimer = 0;
      const roll = Math.random();
      const spawnX = 40 + Math.random() * (W - 80);

      if (roll < 0.5) {
        enemies.push(new Drone(spawnX, -30));
      } else if (roll < 0.8) {
        enemies.push(new Asteroid(spawnX, -40, 26 + Math.random() * 8));
      } else {
        enemies.push(new Cruiser(spawnX, -50));
      }
    }
  }

  // --- BUCLE PRINCIPAL (requestAnimationFrame) ---
  function loop() {
    requestAnimationFrame(loop);

    let shakeX = 0;
    let shakeY = 0;
    if (shakeAmount > 0) {
      shakeX = (Math.random() - 0.5) * shakeAmount;
      shakeY = (Math.random() - 0.5) * shakeAmount;
      shakeAmount *= 0.88;
      if (shakeAmount < 0.5) shakeAmount = 0;
    }

    ctx.save();
    ctx.translate(shakeX, shakeY);

    ctx.fillStyle = '#050713';
    ctx.fillRect(0, 0, W, H);

    // Fondo de estrellas
    ctx.fillStyle = '#ffffff';
    stars.forEach(st => {
      st.y += st.speed;
      if (st.y > H) {
        st.y = 0;
        st.x = Math.random() * W;
      }
      ctx.globalAlpha = st.alpha;
      ctx.fillRect(st.x, st.y, st.size, st.size);
    });
    ctx.globalAlpha = 1;

    // Onda de Choque EMP
    if (shockwaveRadius > 0 && shockwaveRadius < shockwaveMax) {
      shockwaveRadius += 26;
      ctx.save();
      ctx.lineWidth = 8;
      ctx.strokeStyle = `rgba(0, 255, 230, ${1 - shockwaveRadius / shockwaveMax})`;
      ctx.beginPath();
      ctx.arc(player.x, player.y, shockwaveRadius, 0, Math.PI * 2);
      ctx.stroke();
      ctx.restore();
    }

    if (state === 'PLAYING') {
      player.update();
      updateSpawner();

      // Proyectiles del jugador
      for (let i = projectiles.length - 1; i >= 0; i--) {
        const p = projectiles[i];
        p.update();
        p.draw();

        if (p.y < -30 || p.x < -20 || p.x > W + 20) {
          projectiles.splice(i, 1);
          continue;
        }

        let hitEnemy = false;
        for (let j = enemies.length - 1; j >= 0; j--) {
          const en = enemies[j];
          const dist = Math.hypot(p.x - en.x, p.y - en.y);
          const hitRadius = en.isBoss ? 65 : (en.radius || 24);

          if (dist < hitRadius) {
            en.hp -= 20;
            createExplosion(p.x, p.y, '#00ffff', 4);
            hitEnemy = true;

            if (en.hp <= 0) {
              score += en.points;
              floatingTexts.push(new FloatingText(en.x, en.y, `+${en.points}`, '#00ffc8'));
              createExplosion(en.x, en.y, en.isBoss ? '#ff0055' : '#ff7700', en.isBoss ? 45 : 20);
              AudioEngine.explode(en.isBoss);

              if (en.isBoss) bossActive = false;

              if (en instanceof Asteroid && !en.isSub && en.radius > 20) {
                enemies.push(new Asteroid(en.x - 12, en.y, 14, true));
                enemies.push(new Asteroid(en.x + 12, en.y, 14, true));
              }

              if (Math.random() < 0.22) {
                const types = ['WEAPON', 'SHIELD', 'BOMB'];
                const pType = types[Math.floor(Math.random() * types.length)];
                powerups.push(new PowerUp(en.x, en.y, pType));
              }

              enemies.splice(j, 1);
            }
            break;
          }
        }

        if (hitEnemy) {
          projectiles.splice(i, 1);
        }
      }

      // Proyectiles de enemigos
      for (let i = enemyProjectiles.length - 1; i >= 0; i--) {
        const ep = enemyProjectiles[i];
        ep.update();
        ep.draw();

        if (ep.y > H + 30 || ep.x < -20 || ep.x > W + 20) {
          enemyProjectiles.splice(i, 1);
          continue;
        }

        if (Math.hypot(ep.x - player.x, ep.y - player.y) < 22) {
          enemyProjectiles.splice(i, 1);
          player.hit(20);
        }
      }

      // Actualizar enemigos
      for (let i = enemies.length - 1; i >= 0; i--) {
        const en = enemies[i];
        en.update();
        en.draw();

        if (en.y > H + 60) {
          enemies.splice(i, 1);
          continue;
        }

        const hitRadius = en.isBoss ? 55 : (en.radius || 24);
        if (Math.hypot(en.x - player.x, en.y - player.y) < hitRadius + 18) {
          player.hit(35);
          if (!en.isBoss) {
            createExplosion(en.x, en.y, '#ff4400', 16);
            enemies.splice(i, 1);
          }
        }
      }

      // Actualizar powerups
      for (let i = powerups.length - 1; i >= 0; i--) {
        const pu = powerups[i];
        pu.update();
        pu.draw();

        if (pu.y > H + 30) {
          powerups.splice(i, 1);
          continue;
        }

        if (Math.hypot(pu.x - player.x, pu.y - player.y) < 32) {
          AudioEngine.powerup();
          if (pu.type === 'WEAPON') {
            player.weaponLevel = Math.min(3, player.weaponLevel + 1);
            player.weaponTimer = 700;
            floatingTexts.push(new FloatingText(player.x, player.y - 30, '¡LÁSER POTENCIADO!', '#00e5ff'));
          } else if (pu.type === 'SHIELD') {
            player.hp = Math.min(player.maxHp, player.hp + 45);
            floatingTexts.push(new FloatingText(player.x, player.y - 30, '+ESCUDO', '#00ff77'));
          } else if (pu.type === 'BOMB') {
            player.bombs++;
            floatingTexts.push(new FloatingText(player.x, player.y - 30, '+1 BOMBA EMP', '#ffd600'));
          }
          createExplosion(pu.x, pu.y, '#00ffff', 12);
          powerups.splice(i, 1);
        }
      }

      player.draw();

      for (let i = particles.length - 1; i >= 0; i--) {
        const part = particles[i];
        part.update();
        part.draw();
        if (part.life <= 0) particles.splice(i, 1);
      }

      for (let i = floatingTexts.length - 1; i >= 0; i--) {
        const ft = floatingTexts[i];
        ft.update();
        ft.draw();
        if (ft.life <= 0) floatingTexts.splice(i, 1);
      }

      drawHUD();

    } else if (state === 'START') {
      drawStartScreen();
    } else if (state === 'GAMEOVER') {
      drawGameOverScreen();
    }

    ctx.restore();
  }
